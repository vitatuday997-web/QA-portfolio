pwd                         — где я сейчас
dir                         — показать файлы и папки

cd [folder]                 — зайти в папку
cd ..                       — на папку выше
Tab                         — автодополнение имени

mkdir [folder]              — создать папку
New-Item [file.ext]         — создать файл

Rename-Item [old] [new]     — переименовать
Remove-Item [file]          — удалить файл
Remove-Item [folder] -Recurse — удалить папку
Copy-Item [file] [destination] — скопировать файл
Copy-Item [folder] [destination] -Recurse — скопировать папку

code .                      — открыть текущую папку в VS Code
cls                         — очистить терминал
↑                           — предыдущая команда
Ctrl+C                      — отменить текущую команду

git status                  — посмотреть изменения
git add .                   — добавить все изменения
git commit -m "[message]"   — создать коммит
git push                    — отправить на GitHub
git pull                    — забрать измененияremo
